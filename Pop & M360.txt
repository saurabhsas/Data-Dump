CREATE TABLE WORK.MED_CLAIMS_RAW AS

    ADMITDATE,
    DISCHARGEDATE,
    PAID,

-----------------------------------------------------------------------*/

ELSE IF POSDESCR = 'EMERGENCY' THEN CLAIM_TYPE = 'ED';
ELSE CLAIM_TYPE = 'OP';
RUN;

/**************************************************************************
 PHASE 1D: COST & UTILIZATION (HRP, VISITS, READMITS)

/*-----------------------------------------------------------------------
  MEDICAL COST (IP / OP / ED)
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.MED_COST AS
SELECT
    MEMBERID,
    YYMMM,
    SUM(PAID)    AS MED_PAID,
    SUM(ALLOWED) AS MED_ALLOWED
FROM WORK.ALL_CLAIMS
WHERE CLAIM_TYPE IN ('IP','OP','ED')
GROUP BY MEMBERID, YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  PHARMACY COST
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.RX_COST AS
SELECT
    MEMBERID,
    YYMMM,
    SUM(PAID)    AS RX_PAID,
    SUM(ALLOWED) AS RX_ALLOWED
FROM WORK.ALL_CLAIMS
WHERE CLAIM_TYPE = 'RX'
GROUP BY MEMBERID, YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  TOTAL COST (MED + RX)
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.HRP_COST AS
SELECT
    COALESCE(A.MEMBERID, B.MEMBERID) AS MEMBERID,
    COALESCE(A.YYMMM,    B.YYMMM)    AS YYMMM,
    COALESCE(MED_PAID,0) + COALESCE(RX_PAID,0)       AS TOTAL_PAID,
    COALESCE(MED_ALLOWED,0) + COALESCE(RX_ALLOWED,0) AS TOTAL_ALLOWED,
    MED_PAID,
    RX_PAID
FROM WORK.MED_COST A
FULL JOIN WORK.RX_COST B
  ON A.MEMBERID = B.MEMBERID
 AND A.YYMMM    = B.YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  INPATIENT ADMISSIONS & LENGTH OF STAY
-----------------------------------------------------------------------*/
DATA WORK.IP_ADMITS;
SET WORK.ALL_CLAIMS;
WHERE CLAIM_TYPE = 'IP'
  AND ADMITDATE IS NOT MISSING
  AND DISCHARGEDATE IS NOT MISSING;

LOS = DISCHARGEDATE - ADMITDATE;
IF LOS < 0 THEN LOS = .;
RUN;

/*-----------------------------------------------------------------------
  IP ADMISSION COUNTS
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.IP_UTIL AS
SELECT
    MEMBERID,
    YYMMM,
    COUNT(DISTINCT CLAIMID) AS IP_ADMIT_CNT,
    SUM(LOS)                AS IP_DAYS
FROM WORK.IP_ADMITS
GROUP BY MEMBERID, YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  ED VISIT COUNTS
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.ED_UTIL AS
SELECT
    MEMBERID,
    YYMMM,
    COUNT(DISTINCT CLAIMID) AS ED_VISIT_CNT
FROM WORK.ALL_CLAIMS
WHERE CLAIM_TYPE = 'ED'
GROUP BY MEMBERID, YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  OP VISIT COUNTS
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.OP_UTIL AS
SELECT
    MEMBERID,
    YYMMM,
    COUNT(DISTINCT CLAIMID) AS OP_VISIT_CNT
FROM WORK.ALL_CLAIMS
WHERE CLAIM_TYPE = 'OP'
GROUP BY MEMBERID, YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  30-DAY READMISSION LOGIC
-----------------------------------------------------------------------*/
PROC SORT DATA=WORK.IP_ADMITS;
BY MEMBERID ADMITDATE;
RUN;

DATA WORK.READMIT_30;
SET WORK.IP_ADMITS;
BY MEMBERID;

RETAIN PREV_DISCHG;

IF FIRST.MEMBERID THEN PREV_DISCHG = .;

READMIT_30_FLAG = 0;
IF PREV_DISCHG NE . THEN DO;
    IF ADMITDATE - PREV_DISCHG <= 30 THEN READMIT_30_FLAG = 1;
END;

PREV_DISCHG = DISCHARGEDATE;
KEEP MEMBERID YYMMM READMIT_30_FLAG;
RUN;

/*-----------------------------------------------------------------------
  AGGREGATE READMISSION FLAGS
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.READMIT_30_SUM AS
SELECT
    MEMBERID,
    YYMMM,
    MAX(READMIT_30_FLAG) AS READMIT_30_FLAG
FROM WORK.READMIT_30
GROUP BY MEMBERID, YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  FINAL UTILIZATION SUMMARY
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.UTIL_SUMMARY AS
SELECT
    A.MEMBERID,
    A.YYMMM,
    COALESCE(B.IP_ADMIT_CNT,0)   AS IP_ADMIT_CNT,
    COALESCE(B.IP_DAYS,0)        AS IP_DAYS,
    COALESCE(C.ED_VISIT_CNT,0)   AS ED_VISIT_CNT,
    COALESCE(D.OP_VISIT_CNT,0)   AS OP_VISIT_CNT,
    COALESCE(E.READMIT_30_FLAG,0) AS READMIT_30_FLAG
FROM WORK.HRP_COST A
LEFT JOIN WORK.IP_UTIL        B ON A.MEMBERID=B.MEMBERID AND A.YYMMM=B.YYMMM
LEFT JOIN WORK.ED_UTIL        C ON A.MEMBERID=C.MEMBERID AND A.YYMMM=C.YYMMM
LEFT JOIN WORK.OP_UTIL        D ON A.MEMBERID=D.MEMBERID AND A.YYMMM=D.YYMMM
LEFT JOIN WORK.READMIT_30_SUM E ON A.MEMBERID=E.MEMBERID AND A.YYMMM=E.YYMMM;
QUIT;

/*-----------------------------------------------------------------------
  QA CHECK
-----------------------------------------------------------------------*/
PROC SQL;
SELECT
    COUNT(DISTINCT MEMBERID) AS UTIL_MEMBERS
FROM WORK.UTIL_SUMMARY;
QUIT;

/*-----------------------------------------------------------------------
  EDC EXTRACTION
-----------------------------------------------------------------------*/
DATA WORK.EDC_RAW;
SET WORK.ACGDATA_CURRENT;
LENGTH EDC $6;

    EDC = SCAN(EDC_CODES,I,' ');
    OUTPUT;

  EDC TRANSPOSE
-----------------------------------------------------------------------*/
PROC SORT DATA=WORK.EDC_RAW NODUPKEY;
BY MEMBERID EDC;

PROC TRANSPOSE DATA=WORK.EDC_RAW
               OUT=WORK.EDC_T (DROP=_NAME_)
               PREFIX=EDC_;
BY MEMBERID;
VAR EDC;
RUN;

/*-----------------------------------------------------------------------
  CHRONIC CONDITION FLAGS
-----------------------------------------------------------------------*/
DATA WORK.CHRONIC_FLAGS;
SET WORK.EDC_T;

ARRAY EDCV {*} EDC_:;
CHRONIC_CNT = 0;

DO I = 1 TO DIM(EDCV);
    IF NOT MISSING(EDCV[I]) THEN CHRONIC_CNT + 1;
END;

DROP I;
RUN;

/*-----------------------------------------------------------------------
  CANCER FLAGS (POP ORIGINAL ICD LOGIC)
-----------------------------------------------------------------------*/
DATA WORK.CANCER_FLAGS;
SET WORK.ALL_CLAIMS;
LENGTH CANCER_FLAG 3;

IF SUBSTR(ICD10DX1,1,1) = 'C' THEN CANCER_FLAG = 1;
ELSE CANCER_FLAG = 0;

KEEP MEMBERID YYMMM CANCER_FLAG;
RUN;

PROC SQL;
CREATE TABLE WORK.CANCER_SUM AS
SELECT
    MEMBERID,
    MAX(CANCER_FLAG) AS CANCER_FLAG
FROM WORK.CANCER_FLAGS
GROUP BY MEMBERID;
QUIT;

/*-----------------------------------------------------------------------
  BEHAVIORAL HEALTH FLAGS
-----------------------------------------------------------------------*/
DATA WORK.BH_FLAGS;
SET WORK.ALL_CLAIMS;
LENGTH BH_FLAG 3;

IF SUBSTR(ICD10DX1,1,1) IN ('F') THEN BH_FLAG = 1;
ELSE BH_FLAG = 0;

KEEP MEMBERID YYMMM BH_FLAG;
RUN;

PROC SQL;
CREATE TABLE WORK.BH_SUM AS
SELECT
    MEMBERID,
    MAX(BH_FLAG) AS BH_FLAG
FROM WORK.BH_FLAGS
GROUP BY MEMBERID;
QUIT;

  FINAL CLINICAL SUMMARY

QUIT;

/*-----------------------------------------------------------------------

/*-----------------------------------------------------------------------
  JIVA ENGAGEMENT FLAGS
-----------------------------------------------------------------------*/
PROC SQL;
CREATE TABLE WORK.JIVA_FLAGS AS
SELECT
    MEMBERID,
    MAX(CASE WHEN PROGRAM_NAME IS NOT NULL THEN 1 ELSE 0 END) AS DM_FLAG,
    MAX(CASE WHEN EPISODE_NAME IS NOT NULL THEN 1 ELSE 0 END) AS EPISODE_FLAG
FROM
(
