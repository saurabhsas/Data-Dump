
/**************************************************************************
 CHUNK 01: ENVIRONMENT, OPTIONS, MACROS, DATE LOGIC
**************************************************************************/
options compress=yes reuse=yes mprint mlogic symbolgen;

%let path = data;
%let DataSubFolder = php;
%let TableauShare = TSTTBLOAPP01_PHP;
%let TableauShareFP = /&path./&DataSubFolder./&TableauShare.;

filename TXTSRC  "/&TableauShareFP./POP_ASSMNT_SUMMARY.txt";
filename FINSRC  "/&TableauShareFP./POP_ASSMNT_DETAIL.txt";
filename DETAIL  "/&TableauShareFP./POP_ASSMNT_DETAIL2.txt";

%let BUS_UT = 'PHP','PHP/PDS';

data _null_;
    rdate = intnx('month', today(), -1, 'e');
    acgend = intnx('month', rdate, -1, 'e');
    acgbgn = intnx('year', acgend, -1, 's');

    call symputx('CYPM', put(acgend, yymmn6.));
    call symputx('PYPM', put(acgbgn, yymmn6.));
    call symputx('SDATE', put(acgbgn, date9.));
    call symputx('EDATE', put(acgend, date9.));
run;


/**************************************************************************
 CHUNK 02: ENROLLMENT & MEMBER BASE (POP + M360)
**************************************************************************/
proc sql;
connect to odbc
(
  datasrc=IRADW_SAWESCPDSAPP00_UAT_IND
  user="&cdisql_id."
  pw="&cdisql_pw."
);

create table work.member_base as
select distinct
    eligibilityyearandmonth,
    memberid,
    memberuci,
    subscriberid as subscriber_key,
    fsproduct,
    lineofbusiness as cml_flag,
    metalliclevel,
    county,
    memberstate,
    memberterritoryzip,
    rateareaid,
    marketsegment,
    memberlanguage,
    dualstatus,
    dsnpflag,
    vbp_risk_cd,
    riskadjustmentfactor
from IRADW_UAT.DM_M360.CUDA_MM
where eligibilityyearandmonth between &PYPM. and &CYPM.
  and businessunit in (&BUS_UT.);
disconnect from odbc;
quit;

proc sort data=work.member_base;
by memberid descending eligibilityyearandmonth;
run;

data work.member_base;
set work.member_base;
by memberid;
if first.memberid;
drop eligibilityyearandmonth;
run;



/**************************************************************************
 CHUNK 03: CLAIMS – MEDICAL & PHARMACY
**************************************************************************/
proc sql;
connect to odbc
(
  datasrc=IRADW_SAWESCPDSAPP00_UAT_IND
  user="&cdisql_id."
  pw="&cdisql_pw."
);

create table work.med_claims as
select
    memberid,
    claimid,
    servicedate,
    paid,
    allowed,
    posdescr
from IRADW_UAT.DM_M360.CUDA_CLAIMS
where servicedate between "&SDATE."d and "&EDATE."d
  and paid > 0;
disconnect from odbc;
quit;

data work.med_claims;
set work.med_claims;
length claim_type $3;
if posdescr='INPATIENT' then claim_type='IP';
else if posdescr='EMERGENCY' then claim_type='ED';
else claim_type='OP';
run;

proc sql;
connect to odbc
(
  datasrc=IRADW_SAWESCPDSAPP00_UAT_IND
  user="&cdisql_id."
  pw="&cdisql_pw."
);

create table work.rx_claims as
select
    memberid,
    filldate as servicedate,
    paid,
    allowed
from IRADW_UAT.DM_M360.CUDA_RX
where filldate between "&SDATE."d and "&EDATE."d
  and paid > 0;
disconnect from odbc;
quit;

data work.rx_claims;
set work.rx_claims;
claim_type='RX';
run;

data work.all_claims;
set work.med_claims work.rx_claims;
run;



/**************************************************************************
 CHUNK 04: COST & UTILIZATION
**************************************************************************/
proc sql;
create table work.cost_util as
select
    memberid,
    sum(paid)    as total_paid,
    sum(allowed) as total_allowed,
    sum(case when claim_type='RX' then paid else 0 end) as rx_paid,
    sum(case when claim_type ne 'RX' then paid else 0 end) as med_paid
from work.all_claims
group by memberid;
quit;

proc sql;
create table work.utilization as
select
    memberid,
    sum(case when claim_type='IP' then 1 else 0 end) as ip_admit_cnt,
    sum(case when claim_type='ED' then 1 else 0 end) as ed_visit_cnt,
    sum(case when claim_type='OP' then 1 else 0 end) as op_visit_cnt
from work.all_claims
group by memberid;
quit;

/**************************************************************************
 CHUNK 07: SDOH (SOCIAL DETERMINANTS OF HEALTH)
**************************************************************************/
proc sql;
connect to odbc
(
  datasrc=IRADW_SAWESCPDSAPP00_UAT_IND
  user="&cdisql_id."
  pw="&cdisql_pw."
);

create table work.sdoh_raw as
select
    uci_id,
    food_score,
    housing_score,
    transport_score,
    lifestyle_score,
    overall_score,
    race,
    ethnicity,
    score_date
from DM_M360.SDOH_SEF_SCORE;
disconnect from odbc;
quit;

proc sort data=work.sdoh_raw;
by uci_id descending score_date;
run;

data work.sdoh;
set work.sdoh_raw;
by uci_id;
if first.uci_id;

rename
    food_score      = sdoh_food_score
    housing_score   = sdoh_housing_score
    transport_score = sdoh_transport_score
    lifestyle_score = sdoh_lifestyle_score
    overall_score   = sdoh_overall_score
    race            = sdoh_race
    ethnicity       = sdoh_ethnicity;
run;



/**************************************************************************
 CHUNK 08: FINAL ASSEMBLY
**************************************************************************/
proc sql;
create table work.pop_assessment_final as
select
    M.*,
    C.*,
    U.ip_admit_cnt, U.ed_visit_cnt, U.op_visit_cnt,
    E.dm_flag,
    A.*,
    S.sdoh_food_score,
    S.sdoh_housing_score,
    S.sdoh_transport_score,
    S.sdoh_lifestyle_score,
    S.sdoh_overall_score,
    S.sdoh_race,
    S.sdoh_ethnicity
from work.member_base M
left join work.cost_util   C on M.memberid = C.memberid
left join work.utilization U on M.memberid = U.memberid
left join work.engagement  E on M.memberid = E.memberid
left join work.acg         A on M.memberid = A.memberid
left join work.sdoh        S on M.memberuci = S.uci_id;
quit;



/**************************************************************************
 CHUNK 09: TABLEAU OUTPUT FILES
**************************************************************************/
data _null_;
set work.pop_assessment_final;
file TXTSRC dlm='|' lrecl=32767;
put memberid memberuci fsproduct cml_flag county sdoh_overall_score;
run;

data _null_;
set work.pop_assessment_final;
file FINSRC dlm='|' lrecl=32767;
put memberid total_paid total_allowed med_paid rx_paid;
run;

data _null_;
set work.pop_assessment_final;
file DETAIL dlm='|' lrecl=32767;
put memberid sdoh_food_score sdoh_housing_score sdoh_transport_score
    sdoh_lifestyle_score sdoh_overall_score sdoh_race sdoh_ethnicity;
run;



/**************************************************************************
 CHUNK 10: DASHBOARD SCHEMA COMPLETION
**************************************************************************/
data work.pop_dashboard_complete;
set work.pop_assessment_final;

member_months=12;
if member_months>0 then pmpm=total_paid/member_months;

ed_per_1000=(ed_visit_cnt/member_months)*1000;
ip_per_1000=(ip_admit_cnt/member_months)*1000;

high_cost_flag=(total_paid>=50000);
high_util_flag=(ed_visit_cnt>=3 or ip_admit_cnt>=2);
high_risk_flag=(acg_risk>=4);

length risk_tier $12;
if high_cost_flag then risk_tier='VERY HIGH';
else if high_risk_flag then risk_tier='HIGH';
else if acg_risk>=2 then risk_tier='MEDIUM';
else risk_tier='LOW';

run;
