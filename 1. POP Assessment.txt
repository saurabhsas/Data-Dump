       WHEN T1.PROC_CD IN ('H0039')                                 THEN 'Assertive Comm Clinic'
       WHEN T1.PROC_CD IN ('H0015')                                 THEN 'IOP'
       WHEN T1.PROC_CD IN ('H2015')                                 THEN 'CCSS'
       WHEN T1.PROC_CD IN ('H2012')                                 THEN 'Day Treatment'
       WHEN T1.PROC_CD IN ('H2014')                                 THEN 'Behavior Mgmt'
       WHEN T1.PROC_CD IN ('H2017')                                 THEN 'Psychosocial Rehab'
       WHEN T1.PROC_CD IN ('H2033')                                 THEN 'MST'
       WHEN T1.PROC_CD BETWEEN 'T0000' AND 'T9999'                  THEN 'T Codes'
       WHEN T1.PROC_CD IN ('H2022')                                 THEN 'All Faiths WRAP'
       WHEN T1.PROC_CD BETWEEN '80000' AND '89999'
         OR T1.PROC_CD BETWEEN 'G0480' AND 'G0489'                  THEN 'Lab-Primarily Drug'
       WHEN T1.PROC_CD BETWEEN 'G9001' AND 'G9005'                  THEN 'Carelink'
  ELSE 'Other'
  END                    AS COS_PROF
FROM WORK.BH_PROF              T1
RIGHT JOIN WORK.ACG_BH T2 
 ON T1.MEMBERID = T2.MEMBERID
AND T2.YYMMM = &ENDDT.
/*AND (T2.LINE_OF_BUSINESS = '14' OR T2.PRODUCT          = '4' )*/
HAVING COUNT(T1.CLAIMID) NE . AND COS_PROF NE 'Other'
;
QUIT;
