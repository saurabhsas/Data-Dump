import streamlit as st
import pandas as pd
import plotly.express as px
import os
import tempfile
from pptx import Presentation

from core.data_loader import load_data
from core.filters import render_filter_ui, apply_filters_cached
from core.query_router import run_prompt
from core.insights_engine import generate_insights
from core.group_match_loader import load_group_match_data
from core.matching_engine import multi_caliper_matching
from core.matched_data_loader import load_matched_datasets
from core.summary_loader import load_summary_data

# ============================================================
# FORMATTERS
# ============================================================
def format_currency(x):
    try:
        return f"${float(x):,.0f}"
    except:
        return str(x)

def format_number(x):
    try:
        return f"{int(x):,}"
    except:
        return str(x)


# ============================================================
# PPT GENERATOR
# ============================================================
def generate_ppt(matched, k1, k2, insights, data_sample):

    prs = Presentation()

    # Slide 1 - Matching
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Matching Quality"
    slide.placeholders[1].text = f"""
Total Matches: {format_number(len(matched))}
Group1 Members: {format_number(matched['G1_MEMBER_ID'].nunique())}
Group2 Members: {format_number(matched['G2_MEMBER_ID'].nunique())}
"""

    # Slide 2 - KPI
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Key Metrics Overview"

    slide.placeholders[1].text = (
        f"GROUP 1\n"
        f"Members: {format_number(k1['Members'])}\n"
        f"Total Cost: {format_currency(k1['Total Cost'])}\n"
        f"Medical: {format_currency(k1['Medical Cost'])}\n"
        f"Pharmacy: {format_currency(k1['Pharmacy Cost'])}\n"
        f"ED Visits: {format_number(k1['ED Visits'])}\n"
        f"IP Visits: {format_number(k1['IP Visits'])}\n"
        f"PMPM: {format_currency(k1['PMPM'])}\n\n"
        f"GROUP 2\n"
        f"Members: {format_number(k2['Members'])}\n"
        f"Total Cost: {format_currency(k2['Total Cost'])}\n"
        f"Medical: {format_currency(k2['Medical Cost'])}\n"
        f"Pharmacy: {format_currency(k2['Pharmacy Cost'])}\n"
        f"ED Visits: {format_number(k2['ED Visits'])}\n"
        f"IP Visits: {format_number(k2['IP Visits'])}\n"
        f"PMPM: {format_currency(k2['PMPM'])}"
    )

    # Slide 3 - Insights
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Insights"
    slide.placeholders[1].text = "\n".join(insights[:6])

    # Slide 4 - Data Sample
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Data Sample"
    slide.placeholders[1].text = str(data_sample.head(10))

    tmp = tempfile.NamedTemporaryFile(delete=False, suffix=".pptx")
    prs.save(tmp.name)
    return tmp.name


# ============================================================
# APP START
# ============================================================
st.set_page_config(layout="wide")
st.title("🏥 Matched Cohort Analytics Dashboard")

page = st.sidebar.radio(
    "Navigation",
    ["📊 Matched Cohort Dashboard", "📈 Utilization & Cost Comparison"]
)


# ============================================================
# PAGE 1
# ============================================================
if page == "📊 Matched Cohort Dashboard":

    @st.cache_data
    def get_matched():
        g1, g2 = load_group_match_data()
        return multi_caliper_matching(g1, g2)

    df = load_data()
    matched = get_matched()

    # -----------------------------
    # MATCHING PRECISION
    # -----------------------------
    st.markdown("## 🎛 Matching Quality")

    caliper_counts = matched["caliper_used"].value_counts().to_dict()

    CALIPER_DESC = {
        "1e-05": f"Very Strict (Matches: {caliper_counts.get('1e-05',0)})",
        "0.0001": f"Strict (Matches: {caliper_counts.get('0.0001',0)})",
        "0.001": f"Moderate (Matches: {caliper_counts.get('0.001',0)})",
        "0.02": f"Loose (Matches: {caliper_counts.get('0.02',0)})",
        "no_caliper": f"Fallback (Matches: {caliper_counts.get('no_caliper',0)})"
    }

    options = ["ALL"] + list(CALIPER_DESC.keys())

    selected = st.multiselect(
        "Select Matching Precision Levels",
        options=options,
        default=["ALL"]
    )

    filtered_matched = matched if "ALL" in selected else matched[
        matched["caliper_used"].isin(selected)
    ]

    st.markdown("### 📘 Precision Description")
    for k, v in CALIPER_DESC.items():
        st.write(f"{k}: {v}")

    st.write("Total Matches:", len(filtered_matched))

    # -----------------------------
    # LOAD DATA
    # -----------------------------
    g1, g2, _ = load_matched_datasets(df, filtered_matched)
    combined = pd.concat([g1, g2])

    filters = render_filter_ui(combined)
    filtered = apply_filters_cached(combined, filters)

    # -----------------------------
    # KPI
    # -----------------------------
    def compute_kpis(df):
        members = df["MEMBER_ID"].nunique()
        total = df["PAID"].sum()

        return {
            "Members": members,
            "Total Cost": total,
            "Medical Cost": df["MEDICAL_PAID"].sum(),
            "Pharmacy Cost": df["RX_PAID"].sum(),
            "ED Visits": df["EDVISITS"].sum(),
            "IP Visits": df["IPVISITS"].sum(),
            "PMPM": total / members if members else 0
        }

    k1 = compute_kpis(filtered[filtered["GROUP"] == "Group1"])
    k2 = compute_kpis(filtered[filtered["GROUP"] == "Group2"])

    st.markdown("## 📊 Key Metrics Overview")

    col1, col2 = st.columns(2)

    with col1:
        for k, v in k1.items():
            val = format_currency(v) if "Cost" in k or k=="PMPM" else format_number(v)
            st.metric(f"G1 {k}", val)

    with col2:
        for k, v in k2.items():
            val = format_currency(v) if "Cost" in k or k=="PMPM" else format_number(v)
            st.metric(f"G2 {k}", val)

    # -----------------------------
    # ANALYSIS
    # -----------------------------
    st.markdown("## 📈 Analysis")

    selected_prompt = st.selectbox("Select Analysis", PROMPTS)

    result = run_prompt(selected_prompt, filtered)

    if "MONTH" in result.columns:
        result["MONTH"] = result["MONTH"].astype(str).str.replace(".0","")

    fig = build_chart(result, selected_prompt)
    st.plotly_chart(fig, use_container_width=True)

    # -----------------------------
    # INSIGHTS
    # -----------------------------
    insights = generate_insights(selected_prompt, result)

    st.markdown("## 🧠 Insights")
    for ins in insights:
        st.write("•", ins)

    # -----------------------------
    # DATA
    # -----------------------------
    st.markdown("## 📄 Data Sample")
    st.dataframe(result.head(30))

    # -----------------------------
    # PPT EXPORT
    # -----------------------------
    st.markdown("## 📥 Export")

    if st.button("Generate PPT"):
        ppt = generate_ppt(filtered_matched, k1, k2, insights, filtered)

        with open(ppt, "rb") as f:
            st.download_button(
                "Download PPT",
                f,
                "dashboard.pptx",
                "application/vnd.openxmlformats-officedocument.presentationml.presentation"
            )


# ============================================================
# PAGE 2
# ============================================================
elif page == "📈 Utilization & Cost Comparison":

    st.header("📈 Utilization & Cost Comparison")

    df = load_summary_data()

    if df is not None:
        st.dataframe(df.head(50))
