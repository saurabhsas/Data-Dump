def generate_ppt(matched, k1, k2, result, insights, data_sample, summary_df):
    prs = Presentation()

    # -----------------------------
    # SLIDE 1: MATCHING QUALITY
    # -----------------------------
    slide_layout = prs.slides.add_slide(prs.slide_layouts[1])
    title = slide.shapes.title
    content = slide.placeholders[1]

    title.text = "Matching Quality"
    content.text = f"""
Total Matches: {len(matched)}
Group1 Members: {matched['G1_MEMBER_ID'].nunique()}
Group2 Members: {matched['G2_MEMBER_ID'].nunique()}
"""

    # -----------------------------
    # SLIDE 2: KPI
    # -----------------------------
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    title = slide.shapes.title
    content = slide.placeholders[1]

    title.text = "Key Metrics Overview"
    content.text = f"""
Group1 Cost: {k1['Total Cost']}
Group2 Cost: {k2['Total Cost']}
Group1 PMPM: {k1['PMPM']}
Group2 PMPM: {k2['PMPM']}
"""

    # -----------------------------
    # SLIDE 3: ANALYSIS + INSIGHTS
    # -----------------------------
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    title = slide.shapes.title
    content = slide.placeholders[1]

    title.text = "Analysis & Insights"

    insight_text = "\n".join(insights[:5])
    content.text = insight_text

    # -----------------------------
    # SLIDE 4: DATA SAMPLE
    # -----------------------------
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    title = slide.shapes.title
    content = slide.placeholders[1]

    title.text = "Data Sample"
    content.text = str(data_sample.head(10))

    # -----------------------------
    # FUNCTION FOR CATEGORY SLIDES
    # -----------------------------
    def add_category_slide(name, df):
        slide = prs.slides.add_slide(prs.slide_layouts[1])
        slide.shapes.title.text = name
        slide.placeholders[1].text = str(df.head(15))

    # -----------------------------
    # CATEGORY SPLIT
    # -----------------------------
    if summary_df is not None and not summary_df.empty:

        add_category_slide("FIP - Inpatient", summary_df[summary_df["Category"] == "FIP"])
        add_category_slide("FOP - Outpatient", summary_df[summary_df["Category"] == "FOP"])
        add_category_slide("PROF - Professional", summary_df[summary_df["Category"] == "PROF"])
        add_category_slide("OTH - Others", summary_df[summary_df["Category"] == "OTH"])
        add_category_slide("ADDL - Additional", summary_df[summary_df["Category"] == "ADDL"])

    # -----------------------------
    # SAVE FILE
    # -----------------------------
    tmp = tempfile.NamedTemporaryFile(delete=False, suffix=".pptx")
    prs.save(tmp.name)

    return tmp.name
