# core/summary_pipeline.py

import pandas as pd
import streamlit as st
import nzpy
import numpy as np


@st.cache_data(show_spinner=True)
def run_summary_pipeline():

    # -----------------------------------
    # LOAD MATCHED MEMBERS
    # -----------------------------------
    matched_df = pd.read_csv("data/matched_members.csv", dtype=str)

    g1_ids = matched_df["G1_MEMBER_ID"].astype(str).unique().tolist()
    g2_ids = matched_df["G2_MEMBER_ID"].astype(str).unique().tolist()

    def format_ids(ids):
        return "(" + ",".join([f"'{x}'" for x in ids]) + ")"

    g1_ids_sql = format_ids(g1_ids)
    g2_ids_sql = format_ids(g2_ids)

    # -----------------------------------
    # COMMON DB FUNCTION
    # -----------------------------------
    def fetch_claims(member_ids_sql):

        conn = nzpy.connect(
            user='#######',
            password='#######',
            host='npsaas_dev_db',
            port=5480,
            database='IRADW_UAT'
        )

        query_med = f"""
        SELECT DISTINCT
            MEMBERID as MEMBER_ID,
            CLAIMID,
            MR_LINE_DESC1,
            MR_LINE_DESC2,
            MR_LINE_DESC3,
            PAID,
            SVC_STRT_DT,
            MR_ALLOWED
        FROM IRADW_UAT.DM_M360.CUDA_CLAIMS
        WHERE SVC_STRT_DT BETWEEN '20250101' AND '20251231'
        AND CLAIM_STATUS = 'FINAL'
        AND FINAL_PAID_CLAIM = 'Y'
        AND MEMBERID IN {member_ids_sql}
        """

        df_med = pd.read_sql(query_med, conn)

        query_rx = f"""
        SELECT DISTINCT
            MEMBERID as MEMBER_ID,
            CLAIMID,
            MR_LINE_DESC1,
            MR_LINE_DESC2,
            MR_LINE_DESC3,
            PAID,
            SVC_STRT_DT,
            MR_ALLOWED
        FROM IRADW_UAT.DM_M360.CUDA_RX
        WHERE SVC_STRT_DT BETWEEN '20250101' AND '20251231'
        AND CLAIM_STATUS = 'FINAL'
        AND FINAL_PAID_CLAIM = 'Y'
        AND MEMBERID IN {member_ids_sql}
        """

        df_rx = pd.read_sql(query_rx, conn)
        conn.close()

        df = pd.concat([df_med, df_rx], ignore_index=True)

        # -----------------------------------
        # CLEANING
        # -----------------------------------
        df["PAID"] = pd.to_numeric(df["PAID"], errors="coerce")
        df["MR_ALLOWED"] = pd.to_numeric(df["MR_ALLOWED"], errors="coerce")

        df["MR_LINE_DESC1"] = df["MR_LINE_DESC1"].fillna("").astype(str)
        df["MR_LINE_DESC2"] = df["MR_LINE_DESC2"].fillna("").astype(str)
        df["MR_LINE_DESC3"] = df["MR_LINE_DESC3"].fillna("").astype(str)

        # -----------------------------------
        # CATEGORY BUILD
        # -----------------------------------
        df["SVC_CAT"] = (
            df["MR_LINE_DESC2"] + "-" + df["MR_LINE_DESC3"]
        ).str.strip("-").str.strip()

        svc = df["SVC_CAT"].str.upper()
        mr = df["MR_LINE_DESC1"]

        conditions = [
            (mr == "FIP") & svc.str.contains("ALCOHOL"),
            (mr == "FIP") & svc.str.contains("MAT NORM"),
            (mr == "FIP") & svc.str.contains("MAT CSECT"),
            (mr == "FIP") & svc.str.contains("PSYCHIATRIC"),
            (mr == "FIP") & svc.str.contains("NEWBORN"),
            (mr == "FOP") & svc.str.contains("RADIOLOGY"),
            (mr == "PROF"),
            (mr == "OTH"),
            (mr == "ADDL")
        ]

        conditions = [c.fillna(False) for c in conditions]

        choices = [
            "Alcohol and Drug Abuse",
            "Mat Norm Delivery",
            "Mat Csect Delivery",
            "Psychiatric",
            "Well Newborn",
            "Radiology",
            "Professional",
            "Others",
            "Additional"
        ]

        df["SVC_CAT2"] = np.select(conditions, choices, default=df["SVC_CAT"])
        df["MR_LINE_DESC1_FINAL"] = df["MR_LINE_DESC1"]

        summary = (
            df.groupby(["MR_LINE_DESC1_FINAL", "SVC_CAT2"])
            .agg(
                Total_Claim_Count=("CLAIMID", "nunique"),
                Total_Paid_Amount=("PAID", "sum")
            )
            .reset_index()
        )

        return summary

    # -----------------------------------
    # RUN BOTH GROUPS
    # -----------------------------------
    Summary_Group1 = fetch_claims(g1_ids_sql)
    Summary_Group2 = fetch_claims(g2_ids_sql)

    Summary_Group1["GROUP"] = "Group1"
    Summary_Group2["GROUP"] = "Group2"

    return Summary_Group1, Summary_Group2
