# ============================================================
# 🔧 FORMATTERS (FIX HTML ISSUE)
# ============================================================
def format_currency(x):
    try:
        return f"${float(x):,.0f}"
    except:
        return str(x)

def format_number(x):
    try:
        return f"{int(x):,}"
    except:
        return str(x)


# ============================================================
# 📊 PPT GENERATION FUNCTION
# ============================================================
def generate_ppt(matched, k1, k2, result, insights, data_sample, summary_df=None):

    prs = Presentation()

    # -----------------------------
    # Slide 1: Matching Quality
    # -----------------------------
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Matching Quality"

    slide.placeholders[1].text = f"""
Total Matches: {format_number(len(matched))}
Group1 Members: {format_number(matched['G1_MEMBER_ID'].nunique())}
Group2 Members: {format_number(matched['G2_MEMBER_ID'].nunique())}
"""

    # -----------------------------
    # Slide 2: KPI
    # -----------------------------
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Key Metrics Overview"

    slide.placeholders[1].text = f"""
GROUP 1
-------------------
Members: {format_number(k1.get('Members',0))}
Total Cost: {format_currency(k1.get('Total Cost',0))}
PMPM: {format_currency(k1.get('PMPM',0))}

GROUP 2
-------------------
Members: {format_number(k2.get('Members',0))}
Total Cost: {format_currency(k2.get('Total Cost',0))}
PMPM: {format_currency(k2.get('PMPM',0))}
"""

    # -----------------------------
    # Slide 3: Analysis & Insights
    # -----------------------------
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Analysis & Insights"

    slide.placeholders[1].text = "\n".join([str(i) for i in insights[:6]])

    # -----------------------------
    # Slide 4: Data Sample
    # -----------------------------
    slide = prs.slides.add_slide(prs.slide_layouts[1])
    slide.shapes.title.text = "Data Sample"

    slide.placeholders[1].text = str(data_sample.head(10))

    # -----------------------------
    # Category Slides (Optional)
    # -----------------------------
    if summary_df is not None and not summary_df.empty:

        segments = {
            "FIP - Inpatient": "FIP",
            "FOP - Outpatient": "FOP",
            "PROF - Professional": "PROF",
            "OTH - Others": "OTH",
            "ADDL - Additional": "ADDL"
        }

        for title, code in segments.items():
            slide = prs.slides.add_slide(prs.slide_layouts[1])
            slide.shapes.title.text = title

            seg_df = summary_df[summary_df["Category"] == code]
            slide.placeholders[1].text = str(seg_df.head(15))

    tmp = tempfile.NamedTemporaryFile(delete=False, suffix=".pptx")
    prs.save(tmp.name)

    return tmp.name


# ============================================================
# PAGE CONFIG
# ============================================================
st.set_page_config(layout="wide")
st.title("🏥 Matched Cohort Analytics Dashboard")


page = st.sidebar.radio(
    "Navigation",
    ["📊 Matched Cohort Dashboard", "📈 Utilization & Cost Comparison"]
)


# ============================================================
# 📊 PAGE 1
# ============================================================
if page == "📊 Matched Cohort Dashboard":

    @st.cache_data(show_spinner=False)
    def get_matched():
        g1, g2 = load_group_match_data()
        matched = multi_caliper_matching(g1, g2)

        os.makedirs("data", exist_ok=True)
        matched[["G1_MEMBER_ID", "G2_MEMBER_ID"]].to_csv(
            "data/matched_members.csv", index=False
        )
        return matched

    df = load_data()
    matched = get_matched()

    # -----------------------------
    # Matching Filter
    # -----------------------------
    selected = st.multiselect(
        "Select Matching Precision Levels",
        options=["ALL"] + sorted(matched["caliper_used"].unique()),
        default=["ALL"]
    )

    filtered_matched = matched if "ALL" in selected else matched[
        matched["caliper_used"].isin(selected)
    ]

    st.write("Total Matches:", len(filtered_matched))

    # -----------------------------
    # Load Data
    # -----------------------------
    g1_data, g2_data, _ = load_matched_datasets(df, filtered_matched)
    combined = pd.concat([g1_data, g2_data])

    # -----------------------------
    # Filters
    # -----------------------------
    filters = render_filter_ui(combined)
    filtered = apply_filters_cached(combined, filters)

    # -----------------------------
    # KPI
    # -----------------------------
    def compute_kpis(df):
        members = df["MEMBER_ID"].nunique()
        total = df["PAID"].sum()

        return {
            "Members": members,
            "Total Cost": total,
            "PMPM": total / members if members else 0
        }

    k1 = compute_kpis(filtered[filtered["GROUP"] == "Group1"])
    k2 = compute_kpis(filtered[filtered["GROUP"] == "Group2"])

    st.markdown("## 📊 Key Metrics Overview")
    st.write("Group1:", k1)
    st.write("Group2:", k2)

    # -----------------------------
    # Analysis
    # -----------------------------
    st.markdown("## 📈 Analysis")

    selected_prompt = st.selectbox("Select Analysis", PROMPTS)

    result = run_prompt(selected_prompt, filtered)

    fig = build_chart(result, selected_prompt)
    st.plotly_chart(fig, use_container_width=True)

    insights = generate_insights(selected_prompt, result)

    st.markdown("## 🧠 Insights")
    for ins in insights:
        st.write("•", ins)

    # -----------------------------
    # Data Sample
    # -----------------------------
    st.markdown("## 📄 Data Sample")
    st.dataframe(result.head(20))

    # ============================================================
    # 📥 PPT EXPORT (FINAL)
    # ============================================================
    st.markdown("## 📥 Export")

    if st.button("Generate PPT"):

        ppt_file = generate_ppt(
            matched=filtered_matched,
            k1=k1,
            k2=k2,
            result=result,
            insights=insights,
            data_sample=filtered,
            summary_df=None
        )

        with open(ppt_file, "rb") as f:
            st.download_button(
                label="Download PPT",
                data=f,
                file_name="dashboard_export.pptx",
                mime="application/vnd.openxmlformats-officedocument.presentationml.presentation"
            )


# ============================================================
# 📈 PAGE 2
# ============================================================
elif page == "📈 Utilization & Cost Comparison":

    st.header("📈 Utilization & Cost Comparison")

    df = load_summary_data()

    if df is None or df.empty:
        st.warning("No data available")
    else:
        st.dataframe(df.head(50))
