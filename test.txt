proc sql;	
connect to odbc	
(DATASRC=IRADW_SAWESCPDSAPP00_UAT_IND user="&cdisql_id." pw="&cdisql_pw." );		
create table MBRSHP_HCG (COMPRESS=YES) as 	
select distinct 
MAIN.ACCOUNTID AS ACCOUNT_ID,
MAIN.ACCOUNTNAME AS ACCOUNT_NAME,
MAIN.MEMBERID AS MEMBER_ID,
MAIN.MEMBERUCI AS MBR_UCI,
MAIN.ELIGIBILITYYEARANDMONTH AS ELIGIBILITY_YM,
MAIN.FSRESPONSIBILITY AS FS_RESPONSIBILITY,
MAIN.FSPRODUCT AS FS_PRODUCT,
MAIN.POPSUBGROUP1 AS FS_Subgroup_1,
MAIN.COUNTY as Member_County,
1 AS MM

from connection to odbc 
(select distinct
 ACCOUNTID
,ACCOUNTNAME
,MEMBERID
,MEMBERUCI
,ELIGIBILITYYEARANDMONTH
,FSRESPONSIBILITY
,FSPRODUCT
,POPSUBGROUP1
,COUNTY

from IRADW_UAT.P3_ALLPHI.SDAL_MEMBER_ENROLLMENT
WHERE ELIGIBILITYYEARANDMONTH between '202301' and '202312') as main
where MAIN.MEMBERID in (SELECT DISTINCT MEMBER_ID from STAYER )
;		
disconnect from odbc;	
quit;
