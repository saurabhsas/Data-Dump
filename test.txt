# 📅 TIME WINDOW FILTER (FINAL FIX)
# -----------------------------------
st.markdown("### 📅 Select Time Range")

time_range = st.selectbox(
    "Select Period",
    ["Last 3 Months", "Last 6 Months", "Last 12 Months"],
    index=2
)

# -----------------------------------
# USE YYYYMM COLUMN (CRITICAL FIX)
# -----------------------------------
if "ELIGIBILITYYEARMONTH" in filtered.columns:

    # Convert to integer safely
    filtered["ELIGIBILITYYEARMONTH"] = pd.to_numeric(
        filtered["ELIGIBILITYYEARMONTH"], errors="coerce"
    )

    # Get max month
    max_ym = int(filtered["ELIGIBILITYYEARMONTH"].max())

    # Convert YYYYMM → datetime
    max_date = pd.to_datetime(str(max_ym), format="%Y%m")

    # Define window
    if time_range == "Last 3 Months":
        start_date = max_date - pd.DateOffset(months=3)
    elif time_range == "Last 6 Months":
        start_date = max_date - pd.DateOffset(months=6)
    else:
        start_date = max_date - pd.DateOffset(months=12)

    start_ym = int(start_date.strftime("%Y%m"))

    # 🔥 FILTER USING YYYYMM
    filtered = filtered[
        filtered["ELIGIBILITYYEARMONTH"] >= start_ym
    ]

    st.caption(f"Showing data from {start_ym} to {max_ym}")
