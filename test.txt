# core/summary_pipeline.py

import pandas as pd
import streamlit as st
import nzpy
import numpy as np


@st.cache_data(show_spinner=True)
def run_summary_pipeline():

    # -----------------------------------
    # LOAD MATCHED MEMBERS
    # -----------------------------------
    matched_df = pd.read_csv("data/matched_members.csv", dtype=str)

    g1_ids = matched_df["G1_MEMBER_ID"].astype(str).unique().tolist()
    g2_ids = matched_df["G2_MEMBER_ID"].astype(str).unique().tolist()

    def format_ids(ids):
        return "(" + ",".join([f"'{x}'" for x in ids]) + ")"

    g1_ids_sql = format_ids(g1_ids)
    g2_ids_sql = format_ids(g2_ids)

    # -----------------------------------
    # DB CONNECTION
    # -----------------------------------
    def get_conn():
        return nzpy.connect(
            user="XXXXXXXX",
            password="XXXXXXXX",
            host="npsaas_dev_db",
            port=5480,
            database="IRADW_UAT"
        )

    # -----------------------------------
    # COMMON PROCESS
    # -----------------------------------
    def process(member_ids_sql):

        conn = get_conn()

        query = f"""
        SELECT
            MEMBERID AS MEMBER_ID,
            CLAIMID,
            MR_LINE_DESC1,
            MR_LINE_DESC2,
            MR_LINE_DESC3,
            PAID
        FROM IRADW_UAT.DM_M360.CUDA_CLAIMS
        WHERE FINAL_PAID_CLAIM = 'Y'
        AND CLAIM_STATUS = 'FINAL'
        AND MEMBERID IN {member_ids_sql}
        """

        df = pd.read_sql(query, conn)
        conn.close()

        # CLEAN
        df["PAID"] = pd.to_numeric(df["PAID"], errors="coerce")
        df = df.fillna("")

        # BUILD CATEGORY
        df["SVC_CAT"] = (df["MR_LINE_DESC2"] + "-" + df["MR_LINE_DESC3"]).str.strip("-")

        svc = df["SVC_CAT"].str.upper()
        mr = df["MR_LINE_DESC1"]

        # CATEGORY MAPPING
        conditions = [
            (mr == "FIP") & svc.str.contains("MAT NORM"),
            (mr == "FIP") & svc.str.contains("MAT CSECT"),
            (mr == "FOP") & svc.str.contains("RADIOLOGY"),
            (mr == "PROF"),
            (mr == "OTH"),
            (mr == "ADDL")
        ]

        conditions = [c.fillna(False) for c in conditions]

        choices = [
            "Mat Norm Delivery",
            "Mat Csect Delivery",
            "Radiology",
            "Professional",
            "Others",
            "Additional"
        ]

        df["SVC_CAT2"] = np.select(conditions, choices, default=df["SVC_CAT"])

        # FINAL CATEGORY
        df["MR_LINE_DESC1_FINAL"] = df["MR_LINE_DESC1"]

        # CLEAN SUBCATEGORY
        df["SVC_CAT2"] = df["SVC_CAT2"].str.strip("-").str.strip()

        # AGGREGATION
        summary = (
            df.groupby(["MR_LINE_DESC1_FINAL", "SVC_CAT2"])
            .agg(
                Total_Claim_Count=("CLAIMID", "nunique"),
                Total_Paid_Amount=("PAID", "sum")
            )
            .reset_index()
        )

        # -----------------------------------
        # SECTION TOTALS
        # -----------------------------------
        totals = (
            summary.groupby("MR_LINE_DESC1_FINAL")
            .agg(
                Total_Claim_Count=("Total_Claim_Count", "sum"),
                Total_Paid_Amount=("Total_Paid_Amount", "sum")
            )
            .reset_index()
        )

        totals["SVC_CAT2"] = totals["MR_LINE_DESC1_FINAL"] + " TOTAL"

        # APPEND TOTAL ROWS
        summary = pd.concat([summary, totals], ignore_index=True)

        # -----------------------------------
        # ORDERING
        # -----------------------------------
        order = ["FIP", "FOP", "PROF", "OTH", "ADDL"]

        summary["MR_LINE_DESC1_FINAL"] = pd.Categorical(
            summary["MR_LINE_DESC1_FINAL"],
            categories=order,
            ordered=True
        )

        summary = summary.sort_values(
            ["MR_LINE_DESC1_FINAL", "SVC_CAT2"]
        ).reset_index(drop=True)

        summary.index = summary.index + 1

        return summary

    g1 = process(g1_ids_sql)
    g2 = process(g2_ids_sql)

    g1["GROUP"] = "Group1"
    g2["GROUP"] = "Group2"

    return g1, g2
