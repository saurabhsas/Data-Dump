# 📅 TIME WINDOW FILTER (FIXED)

st.markdown("### 📅 Select Time Range")

time_range = st.selectbox(
    "Select Period",
    ["Last 3 Months", "Last 6 Months", "Last 12 Months"],
    index=2
)

if "SVC_STRT_DT" in filtered.columns:

    # Ensure datetime
    filtered["SVC_STRT_DT"] = pd.to_datetime(filtered["SVC_STRT_DT"], errors="coerce")

    # Only consider rows with valid dates (actual claims)
    valid_dates = filtered["SVC_STRT_DT"].dropna()

    if not valid_dates.empty:

        max_date = valid_dates.max()

        if time_range == "Last 3 Months":
            start_date = max_date - pd.DateOffset(months=3)
        elif time_range == "Last 6 Months":
            start_date = max_date - pd.DateOffset(months=6)
        else:
            start_date = max_date - pd.DateOffset(months=12)

        # 🔥 FILTER CLAIMS ONLY (IMPORTANT)
        filtered = filtered[
            (filtered["SVC_STRT_DT"].isna()) |  # keep no-claim members
            (filtered["SVC_STRT_DT"] >= start_date)
        ]

        st.caption(f"Showing data from {start_date.date()} to {max_date.date()}")
