# 📊 PAGE 1 (UNCHANGED)
# ============================================================
if page == "📊 Matched Cohort Dashboard":

    @st.cache_data(show_spinner=False)
    def get_matched():
        g1, g2 = load_group_match_data()
        matched = multi_caliper_matching(g1, g2)

        os.makedirs("data", exist_ok=True)
        matched[["G1_MEMBER_ID", "G2_MEMBER_ID"]].to_csv(
            "data/matched_members.csv",
            index=False
        )
        return matched

    df = load_data()
    matched = get_matched()

    st.markdown("## 🎛 Matching Quality")

    caliper_counts = matched["caliper_used"].value_counts().to_dict()

    CALIPER_DESC = {
        "1e-05": f"Very Strict — Matches: {caliper_counts.get('1e-05',0)}",
        "0.0001": f"Strict — Matches: {caliper_counts.get('0.0001',0)}",
        "0.001": f"Moderate — Matches: {caliper_counts.get('0.001',0)}",
        "0.02": f"Loose — Matches: {caliper_counts.get('0.02',0)}",
        "no_caliper": f"Fallback — Matches: {caliper_counts.get('no_caliper',0)}"
    }

    available_calipers = ["ALL"] + sorted(matched["caliper_used"].unique())

    selected = st.multiselect(
        "Select Matching Precision Levels",
        options=available_calipers,
        default=["ALL"],
        format_func=lambda x: "ALL" if x == "ALL" else f"{x} → {CALIPER_DESC.get(x,'')}"
    )

    filtered_matched = matched if "ALL" in selected else matched[
        matched["caliper_used"].isin(selected)
    ]

    st.markdown("### 📘 Precision Levels Description")
    for k, v in CALIPER_DESC.items():
        st.markdown(f"**{k}** — {v}")

    colA, colB, colC = st.columns(3)
    colA.metric("Total Matches", len(filtered_matched))
    colB.metric("Group1 Members", filtered_matched["G1_MEMBER_ID"].nunique())
    colC.metric("Group2 Members", filtered_matched["G2_MEMBER_ID"].nunique())

    g1_data, g2_data, _ = load_matched_datasets(df, filtered_matched)
    combined = pd.concat([g1_data, g2_data])

    filters = render_filter_ui(combined)
    filtered = apply_filters_cached(combined, filters)

    def compute_kpis(df, matched_df, group):
        members = matched_df["G1_MEMBER_ID"].nunique() if group == "Group1" else matched_df["G2_MEMBER_ID"].nunique()
        total = df["PAID"].sum()
        return {
            "Members": members,
            "Total Cost": total,
            "Medical Cost": df["MEDICAL_PAID"].sum(),
            "Pharmacy Cost": df["RX_PAID"].sum(),
            "ED Visits": df["EDVISITS"].sum(),
            "IP Visits": df["IPVISITS"].sum(),
            "PMPM": total / members if members else 0
        }

    k1 = compute_kpis(filtered[filtered["GROUP"] == "Group1"], filtered_matched, "Group1")
    k2 = compute_kpis(filtered[filtered["GROUP"] == "Group2"], filtered_matched, "Group2")

    st.markdown("## 📊 Key Metrics Overview")

    col1, col2 = st.columns(2)

    def render_kpis(title, kpi):
        st.markdown(f"### {title}")
        cols = st.columns(4)
        for i, (key, val) in enumerate(kpi.items()):
            value = f"${val:,.0f}" if "Cost" in key or key == "PMPM" else f"{int(val):,}"
            cols[i % 4].metric(key, value)

    with col1:
        render_kpis("Group1", k1)
    with col2:
        render_kpis("Group2", k2)

# ============================================================
elif page == "📈 Utilization & Cost Comparison":

    st.header("📈 Utilization & Cost Comparison")

    try:
        df = load_summary_data()
    except Exception as e:
        st.error(f"Error loading summary data: {e}")
        st.stop()

    if df is None or df.empty:
        st.error("No summary data available.")
        st.stop()

    summary = df.groupby(["MR_LINE_DESC1_FINAL", "SVC_CAT2", "GROUP"]).agg({
        "Total_Paid_Amount": "sum",
        "Total_Claim_Count": "sum"
    }).reset_index()

    pivot = summary.pivot(
        index=["MR_LINE_DESC1_FINAL", "SVC_CAT2"],
        columns="GROUP",
        values=["Total_Paid_Amount", "Total_Claim_Count"]
    ).fillna(0)

    pivot.columns = [f"{m}_{g}" for m, g in pivot.columns]
    pivot = pivot.reset_index()

    # DELTA
    def pct_diff(a, b):
        return ((a - b) / b * 100) if b != 0 else 0

    pivot["Cost Δ%"] = pivot.apply(
        lambda x: pct_diff(x["Total_Paid_Amount_Group1"], x["Total_Paid_Amount_Group2"]),
        axis=1
    )

    pivot["Utilization Δ%"] = pivot.apply(
        lambda x: pct_diff(x["Total_Claim_Count_Group1"], x["Total_Claim_Count_Group2"]),
        axis=1
    )

    pivot = pivot.rename(columns={
        "Total_Paid_Amount_Group1": "G1 Paid",
        "Total_Paid_Amount_Group2": "G2 Paid",
        "Total_Claim_Count_Group1": "G1 Claim Count",
        "Total_Claim_Count_Group2": "G2 Claim Count"
    })

    # FORMAT
    pivot["G1 Paid"] = pivot["G1 Paid"].apply(lambda x: f"${x:,.0f}")
    pivot["G2 Paid"] = pivot["G2 Paid"].apply(lambda x: f"${x:,.0f}")
    pivot["G1 Claim Count"] = pivot["G1 Claim Count"].astype(int)
    pivot["G2 Claim Count"] = pivot["G2 Claim Count"].astype(int)
    pivot["Cost Δ%"] = pivot["Cost Δ%"].apply(lambda x: f"{x:+.1f}%")
    pivot["Utilization Δ%"] = pivot["Utilization Δ%"].apply(lambda x: f"{x:+.1f}%")

    pivot.columns = [
        "Category", "Subcategory",
        "G1 Paid", "G2 Paid",
        "G1 Claims", "G2 Claims",
        "Cost Δ%", "Utilization Δ%"
    ]

    # ORDER FIX
    category_order = ["FIP", "FOP", "PROF", "OTH", "ADDL"]
    pivot["Category"] = pd.Categorical(pivot["Category"], categories=category_order, ordered=True)
    pivot = pivot.sort_values(["Category", "Subcategory"])

    segments = {
        "🏥 Inpatient (FIP)": "FIP",
        "🏥 Outpatient (FOP)": "FOP",
        "👨‍⚕️ Professional (PROF)": "PROF",
        "📦 Others (OTH)": "OTH",
        "➕ Additional (ADDL)": "ADDL"
    }

    for title, code in segments.items():
        st.markdown(f"### {title}")

        seg_df = pivot[pivot["Category"] == code].copy()

        if not seg_df.empty:
            seg_df = seg_df.reset_index(drop=True)
            seg_df.index = seg_df.index + 1   # 🔥 FIX: proper numbering
            st.dataframe(seg_df, use_container_width=True)
        else:
            st.info(f"No data available for {title}")
